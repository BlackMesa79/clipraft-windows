//! quick 专用原生材质；仅原生验证模式调用 DWM，雾面托盘不依赖材质能力。
use crate::{
    diag, platform,
    settings::{QuickBackground, Settings, Theme},
};
use anyhow::{bail, Context, Result};
use std::{
    mem::size_of,
    sync::atomic::{AtomicIsize, Ordering},
};
use windows::Win32::{Foundation::HWND, Graphics::Dwm::*};

mod surface;

static ACTIVE_WINDOW: AtomicIsize = AtomicIsize::new(0);

#[derive(Clone, Copy)]
pub struct Appearance {
    background: QuickBackground,
    theme: Theme,
}
impl From<&Settings> for Appearance {
    fn from(settings: &Settings) -> Self {
        Self {
            background: settings.quick_background,
            theme: settings.theme,
        }
    }
}

fn set(window: HWND, attribute: DWMWINDOWATTRIBUTE, value: i32) -> Result<()> {
    unsafe {
        DwmSetWindowAttribute(
            window,
            attribute,
            (&value as *const i32).cast(),
            size_of::<i32>() as u32,
        )?;
    }
    Ok(())
}

fn backdrop(window: HWND) -> Result<i32> {
    let mut value = 0_i32;
    unsafe {
        DwmGetWindowAttribute(
            window,
            DWMWA_SYSTEMBACKDROP_TYPE,
            (&mut value as *mut i32).cast(),
            size_of::<i32>() as u32,
        )?;
    }
    Ok(value)
}

fn apply(raw: isize, appearance: Appearance) -> Result<()> {
    clear(raw)?;
    if appearance.background != QuickBackground::Native {
        return Ok(());
    }
    let window = platform::hwnd(raw);
    // 先关闭系统 Acrylic，避免继续显示上一版的整块实色底。
    set(window, DWMWA_SYSTEMBACKDROP_TYPE, DWMSBT_NONE.0)?;
    set(window, DWMWA_USE_HOSTBACKDROPBRUSH, 1)
        .context("原生 Host Backdrop 不可用，请切换为雾面托盘")?;
    ACTIVE_WINDOW.store(raw, Ordering::Release);
    surface::create(raw, appearance.theme).context("创建原生背景画笔失败，请切换为雾面托盘")?;
    if backdrop(window)? != DWMSBT_NONE.0 {
        bail!("系统 Acrylic 底层未成功关闭");
    }
    Ok(())
}

pub fn clear(raw: isize) -> Result<()> {
    if raw == 0 || ACTIVE_WINDOW.load(Ordering::Acquire) != raw {
        return Ok(());
    }
    let window = platform::hwnd(raw);
    surface::clear();
    set(window, DWMWA_USE_HOSTBACKDROPBRUSH, 0)?;
    set(window, DWMWA_SYSTEMBACKDROP_TYPE, DWMSBT_NONE.0)?;
    ACTIVE_WINDOW.store(0, Ordering::Release);
    // 取消材质成功后，装饰属性恢复失败不能阻止用户回到雾面托盘。
    for (attribute, value) in [
        (DWMWA_WINDOW_CORNER_PREFERENCE, DWMWCP_DEFAULT.0),
        (DWMWA_USE_IMMERSIVE_DARK_MODE, 0),
    ] {
        if let Err(error) = set(window, attribute, value) {
            diag::write(format!("quick material decoration reset failed: {error:#}"));
        }
    }
    Ok(())
}

pub fn hide(raw: isize) {
    platform::hide(raw);
    if let Err(error) = clear(raw) {
        diag::write(format!("quick material cleanup failed: {error:#}"));
    }
}

pub fn show(raw: isize, target: isize, size: (f64, f64), appearance: Appearance) -> Result<()> {
    let result = apply(raw, appearance)
        .and_then(|()| platform::show_quick(raw, target, size))
        .and_then(|()| resize(raw));
    if result.is_err() {
        hide(raw);
    }
    result
}

pub fn resize(raw: isize) -> Result<()> {
    if raw != 0 && ACTIVE_WINDOW.load(Ordering::Acquire) == raw {
        surface::resize(raw)?;
    }
    Ok(())
}
