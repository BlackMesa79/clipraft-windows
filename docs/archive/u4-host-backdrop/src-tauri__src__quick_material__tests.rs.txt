use super::*;
use windows::{core::w, Win32::UI::WindowsAndMessaging::*};

#[test]
fn legacy_modes_do_not_require_dwm_on_an_idle_handle() {
    apply(
        0,
        Appearance {
            background: QuickBackground::Frosted,
            theme: Theme::Dark,
        },
    )
    .unwrap();
    clear(0).unwrap();
}

#[test]
fn invalid_native_handle_reports_failure_without_poisoning_legacy_modes() {
    assert!(apply(
        0,
        Appearance {
            background: QuickBackground::Native,
            theme: Theme::Light
        }
    )
    .is_err());
    apply(
        0,
        Appearance {
            background: QuickBackground::Frosted,
            theme: Theme::Light,
        },
    )
    .unwrap();
}

struct HiddenWindow(HWND);
impl Drop for HiddenWindow {
    fn drop(&mut self) {
        hide(self.0 .0 as isize);
        if let Err(error) = unsafe { DestroyWindow(self.0) } {
            diag::write(format!("native probe destroy failed: {error}"));
        }
    }
}

#[test]
#[ignore = "Windows 11 原生隐藏窗口属性探测；显式运行，不代表视觉通过"]
fn native_material_hidden_probe() {
    let styles = WS_EX_NOACTIVATE
        | WS_EX_TRANSPARENT
        | WS_EX_TOOLWINDOW
        | WS_EX_LAYERED
        | WS_EX_NOREDIRECTIONBITMAP;
    let window = HiddenWindow(unsafe {
        CreateWindowExW(
            styles,
            w!("STATIC"),
            w!("ClipRaft hidden material probe"),
            WS_POPUP,
            0,
            0,
            420,
            408,
            None,
            None,
            None,
            None,
        )
        .unwrap()
    });
    unsafe {
        SetLayeredWindowAttributes(
            window.0,
            windows::Win32::Foundation::COLORREF(0),
            255,
            LWA_ALPHA,
        )
        .unwrap();
    }
    let raw = window.0 .0 as isize;
    for theme in [Theme::Light, Theme::Dark] {
        apply(
            raw,
            Appearance {
                background: QuickBackground::Native,
                theme,
            },
        )
        .unwrap();
        assert_eq!(backdrop(window.0).unwrap(), DWMSBT_NONE.0);
        assert!(surface::active(raw));
        assert!(!unsafe { IsWindowVisible(window.0) }.as_bool());
        assert_ne!(unsafe { GetForegroundWindow() }, window.0);
        assert_eq!(
            unsafe { GetWindowLongPtrW(window.0, GWL_EXSTYLE) } as u32 & styles.0,
            styles.0
        );
        resize(raw).unwrap();
        clear(raw).unwrap();
        assert!(!surface::active(raw));
        assert_eq!(backdrop(window.0).unwrap(), DWMSBT_NONE.0);
    }
    apply(
        raw,
        Appearance {
            background: QuickBackground::Native,
            theme: Theme::Dark,
        },
    )
    .unwrap();
    apply(
        raw,
        Appearance {
            background: QuickBackground::Frosted,
            theme: Theme::Dark,
        },
    )
    .unwrap();
    assert_eq!(backdrop(window.0).unwrap(), DWMSBT_NONE.0);
    println!("host_backdrop_graph=created system_acrylic=disabled cleanup=passed hidden=true visual_verified=false");
}
